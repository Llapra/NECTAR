# ---------------------------------------------------------------------------
# NECTAR · dependencias
#
#   pip install -r requirements.txt
#
# Comprobar qué hay instalado y qué falta:
#
#   python main.py --check
#
# El programa arranca con solo PyQt6 y numpy: cada dependencia que falte
# desactiva su función y lo avisa, en vez de impedir el arranque.
# ---------------------------------------------------------------------------

# --- obligatorias ----------------------------------------------------------
# PyQt6 6.7.1 y no una versión más nueva: con el Python de Anaconda, las
# versiones 6.8+ de Qt no cargan, porque el runtime de MSVC que Anaconda trae
# en su carpeta raíz (14.42) es anterior al que necesitan y gana en el orden
# de búsqueda de DLL frente al del sistema (14.44). El síntoma es
# "ImportError: DLL load failed while importing QtCore".
PyQt6==6.7.1
PyQt6-Qt6==6.7.1
numpy>=1.24

# --- aceleración numérica --------------------------------------------------
# Compila a código máquina el renderizado y la integración de eventos, y
# libera el GIL mientras corren. Sin numba el programa funciona, pero el
# procesamiento de eventos usa un camino vectorizado más lento.
numba>=0.58

# --- visualización ---------------------------------------------------------
# Visores de imagen con ROI arrastrable y gráficos en vivo.
pyqtgraph>=0.13.3
# Gráficos de análisis (conversor IMU y topografía).
matplotlib>=3.7
# Grabación del video de la reconstrucción.
# >=4.5 y no >=4.8: OpenEB 5.3 (Metavision libre) fija opencv-python==4.5.5.64
# en sus requisitos de Python. NECTAR solo usa VideoWriter, cvtColor y resize,
# que funcionan igual en 4.5, así que se acepta la versión que trae OpenEB y
# pip no la reemplaza.
opencv-python>=4.5

# --- SDK de la cámara de eventos -------------------------------------------
# Metavision para Windows no instala sus dependencias de Python, y
# metavision_core.event_io importa h5py al cargarse: sin él la cámara de
# eventos no se detecta aunque el SDK esté instalado.
h5py>=3.8

# --- hardware --------------------------------------------------------------
# Puertos serie del IMU/GPS y del anemómetro.
pyserial>=3.5
# Enlace con EmergentCameraC.dll (eSDK de la cámara Emergent).
cffi>=1.15

# --- imágenes --------------------------------------------------------------
tifffile>=2023.7.10
pillow>=10.0

# --- exportación de datos --------------------------------------------------
# .mat en el conversor IMU.
scipy>=1.10
# .parquet en el conversor IMU.
pandas>=2.0
pyarrow>=14.0

# --- pestaña de topografía -------------------------------------------------
rasterio>=1.3
pyproj>=3.5
shapely>=2.0

# ---------------------------------------------------------------------------
# NO se instalan con pip: son SDK del fabricante (ver README, sección 2).
#
#   * Cámara de eventos SilkyEvCam: OpenEB (parte libre de Metavision) +
#     plugin de CenturyArks. Provee metavision_core, metavision_hal y
#     metavision_sdk_core.
#       - Ubuntu 24.04: paquete apt `metavision-openeb` (5.3) + plugin
#         SilkyEvCam v5.2.0. El entorno virtual se crea con
#         --system-site-packages y ANTES de este archivo se instala
#         /usr/share/metavision/python_requirements/requirements_openeb.txt
#       - Windows: instalador de Metavision + plugin de CenturyArks.
#     Sin él, las cámaras de eventos solo funcionan en modo simulado.
#
#   * eSDK de Emergent Vision Technologies — cámara HB-1800-S-M.
#       - Windows: eSDK 2.49 en C:\EVT\eSDK y variable EMERGENT_DIR=C:\EVT
#       - Linux: eSDK para Ubuntu 24.04 / kernel 6.17 (se instala en /opt/EVT)
#     Sin él, las cámaras Emergent solo funcionan en modo demostración.
# ---------------------------------------------------------------------------
